## 当时采用的流程图

```mermaid
flowchart TB
    subgraph S1["第一阶段：系统通电"]
        A["打开 AI·OPS DECK"] --> B["自动检查系统、网络与本机环境"]
        B --> C["离线护航引导启动"]
        C --> D{"是否已有可用模型服务"}
    end

    subgraph S2["第二阶段：接通能源"]
        D -->|"没有"| E["推荐 DeepSeek 快速接入"]
        E --> F["说明用途并跳转申请页面"]
        F --> G["保留当前进度，等待用户返回"]
        G --> H["填写 DeepSeek API Key"]
        D -->|"已有"| H
        H --> I["保存到系统安全钥匙串"]
        I --> J["护航 AI 直接测试连接"]
        J --> K{"连接是否成功"}
        K -->|"失败"| L["解释错误并给出修复步骤"]
        L --> H
        K -->|"成功"| M["AI 护航系统上线"]
    end

    subgraph S3["第三阶段：第一次点火"]
        M --> N["扫描本机 Agent"]
        N --> O{"Pi 是否已经安装"}
        O -->|"没有"| P["解释安装内容并请求用户确认"]
        P --> Q["安装 Pi Agent"]
        O -->|"已经安装"| R["检测 Pi 状态"]
        Q --> R
        R --> S["使用同一个 DeepSeek Key 配置 Pi"]
        S --> T["启动并执行健康检查"]
        T --> U{"Pi 是否可用"}
        U -->|"失败"| V["护航 AI 诊断"]
        V --> W["修复配置 / 重新安装 / 更换方案"]
        W --> R
        U -->|"成功"| X["第一次点火成功"]
    end

    subgraph S4["第四阶段：地面测试"]
        X --> Y["进入内置隔离测试区"]
        Y --> Z["执行安全测试任务"]
        Z --> ZA{"指令、响应、状态是否正常"}
        ZA -->|"失败"| V
        ZA -->|"正常"| ZB["地面测试通过"]
    end

    subgraph S5["第五阶段：第一次试飞"]
        ZB --> ZC{"用户是否有项目"}
        ZC -->|"有"| ZD["选择或导入真实项目"]
        ZC -->|"没有"| ZE["加载内置演示项目"]
        ZD --> ZF["授予只读访问权限"]
        ZE --> ZF
        ZF --> ZG["护航 AI 生成试飞计划"]
        ZG --> ZH["用户确认起飞"]
        ZH --> ZI["Pi 只读梳理项目"]
        ZI --> ZJ["驾驶舱实时展示进度、日志与状态"]
        ZJ --> ZK["生成项目说明和发现报告"]
        ZK --> ZL{"用户是否认可结果"}
        ZL -->|"需要补充"| ZM["补充要求并继续分析"]
        ZM --> ZI
        ZL -->|"认可"| ZN["第一次试飞完成并安全降落"]
        ZN --> ZO["保存报告和黑匣子记录"]
        ZO --> ZP["进入正式驾驶舱"]
    end
```
